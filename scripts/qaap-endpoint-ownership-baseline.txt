# Endpoints in packages/qaap-* that register a route without an auth/ownership primitive
# visible to scripts/qaap-endpoint-ownership-check.js (single-file, in-file delegation only).
# This list may only shrink: add the guard or make it visible to the checker, then delete the line.
# Each entry carries a reviewed reason why the route is intentionally public or guarded off-file.
#
# Regenerate only with review: node scripts/qaap-endpoint-ownership-check.js --write

packages/qaap-cloud-workspace/src/node/qaap-billing-endpoint.ts :: POST QAAP_BILLING_WEBHOOK_API_PATH
# Stripe webhook: authenticated by Stripe's HMAC signature over the raw body
# (verifyStripeWebhookEvent, qaap-stripe-billing.ts), not by session. A session guard
# here would break Stripe's own calls. Auth in handler: none by design.

packages/qaap-shared-core/src/node/qaap-dev-preview-endpoint-render.ts :: POST QAAP_DEV_PREVIEW_CLAIM_PATH
packages/qaap-shared-core/src/node/qaap-dev-preview-endpoint-render.ts :: POST QAAP_DEV_PREVIEW_RELEASE_PATH
# Guard lives in an imported extracted function (handleClaimExtracted /
# handleReleaseExtracted, qaap-dev-preview-endpoint-render.ts / -streaming.ts), which
# calls ctx.auth.authenticate(req) and returns 401 on unauthorized. Off-file, invisible
# to the single-file scan by design of the *Extracted refactor; verified manually.

packages/qaap-shared-core/src/node/qaap-github-oauth-endpoint.ts :: GET QAAP_GITHUB_OAUTH_START_PATH
# OAuth dance entry point: by definition pre-login (it initiates the GitHub OAuth flow).
# Stateless; the state parameter's CSRF protection and the callback validation happen
# later in the flow (handleOAuthCallback is itself guarded).

packages/qaap-cloud-workspace/src/node/qaap-task-sandbox-broker-client.ts :: GET `${QAAP_TENANT_RUNTIME_API_PATH}${QAAP_TASK_SANDBOX_LINK_PATH}`
packages/qaap-cloud-workspace/src/node/qaap-task-sandbox-broker-client.ts :: POST `${QAAP_TENANT_RUNTIME_API_PATH}${QAAP_TASK_SANDBOX_REPLY_PATH}`
# Task-sandbox broker link (doc/qaap-task-sandbox.md, broker mode): machine-to-machine, called only
# by the control plane's QaapTaskSandboxBroker, never by a user session. Registered only in a
# tenant backend with QAAP_TASK_SANDBOX=1 (QaapTaskSandboxBrokerClient.isEnabled). Both handlers
# first call authorized(), i.e. QaapTaskSandboxBrokerAuth.verifyLinkProbe: an HMAC-SHA256 probe over
# the tenant login and a timestamp (60 s window) keyed by this tenant's QAAP_TENANT_BACKEND_SECRET,
# compared in constant time; anything else gets 404. Same trust model as the /busy probe
# (QaapTenantBusyProbe). Verified by "refuses a link without the tenant-scoped probe" in
# qaap-task-sandbox-broker.spec.ts.

packages/qaap-cloud-workspace/src/node/qaap-tenant-runtime-endpoint.ts :: GET `${QAAP_TENANT_RUNTIME_API_PATH}/drain-status`
packages/qaap-cloud-workspace/src/node/qaap-tenant-runtime-endpoint.ts :: POST `${QAAP_TENANT_RUNTIME_API_PATH}/drain`
# Deploy-only drain used by scripts/qaap-vps-update.sh via `docker compose exec theia` over the
# container's own loopback. isLocalDeployRequest() answers 404 unless the peer is loopback AND no
# X-Forwarded-For / X-Forwarded-Host / Forwarded header is present AND the x-qaap-deploy-token
# header matches the per-process token in a 0600 file only the backend user can read, so anything
# through Caddy or the tenant proxy, and agents sharing the netns under their own uid, are refused. No user session exists in that context by design. Verified: public
# GET /qaap/api/cloud/runtime/drain-status returns 404.
